rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Colecciones compartidas (solo lectura para quien tenga el enlace). Se pueden crear y borrar, no cambiar.
    match /shares/{id} {
      allow get: if true;   // abrir un enlace concreto; listar todos no está permitido
      allow create: if id.matches('[A-Za-z0-9]{20,40}') && request.resource.data.items is list
                    && request.resource.data.items.size() <= 20000;
      allow delete: if true;
    }
    match /vaults/{code} {
      allow read, write: if code.matches('TCG-[A-Z0-9]{6}');
      match /{document=**} {
        allow read, write: if code.matches('TCG-[A-Z0-9]{6}');
      }
    }
  }
}
