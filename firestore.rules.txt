rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /vaults/{code} {
      allow read, write: if code.matches('TCG-[A-Z0-9]{6}');
      match /{document=**} {
        allow read, write: if code.matches('TCG-[A-Z0-9]{6}');
      }
    }
  }
}
